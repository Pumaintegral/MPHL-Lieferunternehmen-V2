    function applyFilters() {

      const searchQuery = document.getElementById('searchInput').value.toLowerCase().trim();

      const selectCat = document.getElementById('searchCategory').value;

      const maxPrice = parseInt(document.getElementById('priceRange').value);

      const sortBy = document.getElementById('sortSelect').value;

      

      const ratingRadios = document.getElementsByName('ratingFilter');

      let minRating = 0;

      ratingRadios.forEach(r => { if(r.checked) minRating = parseFloat(r.value); });


      if (selectCat !== 'All' && selectCat !== activeCategory) {

        activeCategory = selectCat;

        renderCategorySidebar();

      }


      let filtered = MINECRAFT_ITEMS.filter(item => {

        const matchesSearch = item.name.toLowerCase().includes(searchQuery) || item.description.toLowerCase().includes(searchQuery);

        const matchesCat = (activeCategory === 'All') || (item.category === activeCategory);

        const matchesPrice = item.price <= maxPrice;

        const matchesRating = item.rating >= minRating;

        return matchesSearch && matchesCat && matchesPrice && matchesRating;

      });


      if (sortBy === 'price-low') filtered.sort((a,b) => a.price - b.price);

      if (sortBy === 'price-high') filtered.sort((a,b) => b.price - a.price);

      if (sortBy === 'rating') filtered.sort((a,b) => b.rating - a.rating);

      if (sortBy === 'name') filtered.sort((a,b) => a.name.localeCompare(b.name));


      document.getElementById('resultsCount').innerText = `${filtered.length} Items gefunden`;

      renderProductsGrid(filtered);

    }

    function adjustQuantity(id, delta) {

    async function processOrderToFirestore() {

      const db = window.firebaseDb;

      const appId = window.appId;


      const totalPrice = cart.reduce((sum, i) => sum + (i.price * i.quantity), 0);

      const generatedId = 'ORD-' + Math.random().toString(36).substring(2, 8).toUpperCase();


      const newOrder = {

        id: generatedId,

        customerName: currentUser.name,

        customerUid: currentUser.uid || 'anon',

        items: cart.map(i => ({ name: i.name, quantity: i.quantity, price: i.price })),

        totalDiamonds: totalPrice,

        status: 'Offen',

        timestamp: Date.now(),

        dateStr: new Date().toLocaleString('de-DE')

      };


      if (db) {

        try {

          const ordersCol = window.fStore.collection(db, 'artifacts', appId, 'public', 'data', 'orders');

          await window.fStore.addDoc(ordersCol, newOrder);

        } catch (e) {

          console.warn("Saving directly to local storage.", e);

        }

      }


      liveOrders = saveLocalOrder(newOrder);

      renderAdminDashboard();


      cart = [];

      updateCartUI();

      closeCheckoutModal();

      showToast(' Bestätigt! Deine Bestellung wurde an den Owner übermittelt.');

    }


    // --- OWNER ADMIN DASHBOARD RENDER ---

    function renderAdminDashboard() {

      const tbody = document.getElementById('adminOrdersTableBody');

      let totalRev = liveOrders.reduce((sum, o) => sum + (o.totalDiamonds || 0), 0);

      let pendingCount = liveOrders.filter(o => o.status === 'Offen').length;


      document.getElementById('adminTotalRevenue').innerText = totalRev;

      document.getElementById('adminTotalOrders').innerText = liveOrders.length;

      document.getElementById('adminPendingOrders').innerText = pendingCount;


      if (liveOrders.length === 0) {

        tbody.innerHTML = `<tr><td colspan="6" class="p-8 text-center text-gray-500">Noch keine Kundenbestellungen eingegangen.</td></tr>`;

        return;

      }


      tbody.innerHTML = liveOrders.map(order => `

        <tr class="hover:bg-gray-750 transition">

          <td class="p-3 font-mono text-[11px]">

            <div class="text-white font-bold">#${order.id.substring(0,8)}</div>

            <div class="text-gray-400 text-[10px]">${order.dateStr || ''}</div>

          </td>

          <td class="p-3 font-bold text-amazon-yellow">

            👤 ${order.customerName || 'Unbekannt'}

          </td>

          <td class="p-3">

            <div class="space-y-0.5 max-h-16 overflow-y-auto">

              ${(order.items || []).map(i => `<span class="inline-block bg-gray-900 px-1.5 py-0.5 rounded text-[10px] mr-1 mb-1 border border-gray-700">${i.name} (x${i.quantity})</span>`).join('')}

            </div>

          </td>

          <td class="p-3 font-pixel text-sm text-mc-diamond font-bold">

            💎 ${order.totalDiamonds}

          </td>

          <td class="p-3">

            ${order.status === 'Offen' 

              ? '<span class="bg-amber-950 text-amber-400 border border-amber-600/50 px-2 py-0.5 rounded text-[10px] font-bold">⏳ Offen</span>' 

              : '<span class="bg-emerald-950 text-emerald-400 border border-emerald-600/50 px-2 py-0.5 rounded text-[10px] font-bold">✅ Geliefert</span>'}

          </td>

          <td class="p-3 text-right space-x-1">

            <button onclick="toggleOrderStatus('${order.id}', '${order.status}')" class="bg-gray-700 hover:bg-gray-600 text-white px-2 py-1 rounded text-[10px] font-bold transition">

              ${order.status === 'Offen' ? 'Als Geliefert' : 'Als Offen'}

            </button>

            <button onclick="deleteOrder('${order.id}')" class="bg-red-900/60 hover:bg-red-800 text-red-300 px-2 py-1 rounded text-[10px] font-bold transition">

              <i class="fa-solid fa-trash"></i>

            </button>

          </td>

        </tr>

      `).join('');

    }


    async function toggleOrderStatus(orderId, currentStatus) {

      const db = window.firebaseDb;

      const appId = window.appId;

      const newStatus = currentStatus === 'Offen' ? 'Geliefert' : 'Offen';


      liveOrders = updateLocalOrderStatus(orderId, newStatus);

      renderAdminDashboard();


      if (db) {

        try {

          const orderRef = window.fStore.doc(db, 'artifacts', appId, 'public', 'data', 'orders', orderId);

          await window.fStore.updateDoc(orderRef, { status: newStatus });

        } catch (e) {}

      }

      showToast(`Status geändert: ${newStatus}`);

    }


    async function deleteOrder(orderId) {

      const db = window.firebaseDb;

      const appId = window.appId;


      liveOrders = deleteLocalOrder(orderId);

      renderAdminDashboard();


      if (db) {

        try {

          const orderRef = window.fStore.doc(db, 'artifacts', appId, 'public', 'data', 'orders', orderId);

          await window.fStore.deleteDoc(orderRef);

        } catch (e) {}

      }

      showToast('Bestellung gelöscht');

    }

      currentUser.isOwner = false;

      document.getElementById('ownerDashboardBtn').classList.add('hidden');


      closeAuthModal();

      showToast(`Angemeldet als ${val}`);

    }

      document.getElementById('headerUserLocation').innerText = 'Admin HQ';

      document.getElementById('ownerDashboardBtn').classList.remove('hidden');

      closeAuthModal();

      showToast('👑 Owner-Zugang erfolgreich aktiviert!');

      switchView('admin');

    }
